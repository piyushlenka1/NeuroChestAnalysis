# --- Define tiers FIRST (used as default args below) ---
TIER1_RELIABLE   = ["Atelectasis", "Cardiomegaly", "Effusion", "Mass",
                     "Pneumothorax", "Edema", "Emphysema", "Hernia"]
TIER2_MARGINAL   = ["Consolidation", "Fibrosis", "Pleural_Thickening"]
TIER3_WEAK       = ["Infiltration", "Nodule", "Pneumonia"]
DETECTION_ELIGIBLE = TIER1_RELIABLE + TIER2_MARGINAL

def get_detection_labels(probs_single, label_cols, final_thresholds,
                         detection_eligible=DETECTION_ELIGIBLE):
    positives = []
    for i, label in enumerate(label_cols):
        if label == "No Finding":
            continue
        if label not in detection_eligible:
            continue
        if probs_single[i] >= final_thresholds[label]:
            positives.append((label, float(probs_single[i])))
    positives.sort(key=lambda x: -x[1])
    return positives

# --- Youden threshold search ---
MIN_PRECISION_FLOOR = 0.10

def find_thresholds_youden(probs, targets, label_cols, min_precision=0.10):
    thresholds  = {}
    diagnostics = {}
    for i, label in enumerate(label_cols):
        y_true = targets[:, i]
        y_prob = probs[:, i]
        if len(np.unique(y_true)) < 2 or y_true.sum() < 5:
            thresholds[label]  = 0.5
            diagnostics[label] = {"method": "fallback", "note": "too few positives"}
            continue
        fpr, tpr, roc_thresh = roc_curve(y_true, y_prob)
        best_j_idx    = np.argmax(tpr - fpr)
        youden_thresh = float(roc_thresh[best_j_idx])
        prec_j = precision_score(y_true, (y_prob >= youden_thresh).astype(int), zero_division=0)
        if prec_j >= min_precision:
            thresholds[label]  = youden_thresh
            diagnostics[label] = {"method": "youden_j",  "prec": prec_j,
                                   "tpr": tpr[best_j_idx], "fpr": fpr[best_j_idx]}
        else:
            precision_arr, recall_arr, pr_thresh = precision_recall_curve(y_true, y_prob)
            p, r = precision_arr[:-1], recall_arr[:-1]
            valid = p >= min_precision
            if valid.any():
                best_idx = np.where(valid, r, -1).argmax()
                thresholds[label]  = float(pr_thresh[best_idx])
                diagnostics[label] = {"method": "precision_floor",
                                       "prec": p[best_idx], "rec": r[best_idx]}
            else:
                with np.errstate(invalid='ignore'):
                    f1_vals = np.where((p + r) > 0, 2 * p * r / (p + r), 0.0)
                thresholds[label]  = float(pr_thresh[np.argmax(f1_vals)])
                diagnostics[label] = {"method": "f1_fallback", "note": "prec floor unreachable"}
    return thresholds, diagnostics

CAM_THRESHOLDS = {
    # Diffuse: spreads across lobe
    "Pneumonia":          0.40,
    "Edema":              0.40,
    "Infiltration":       0.42,
    "Consolidation":      0.42,
    # Focal: tight region
    "Nodule":             0.60,
    "Mass":               0.58,
    "Pneumothorax":       0.62,
    "Emphysema":          0.52,
    # Large structure
    "Cardiomegaly":       0.33,
    # Mixed
    "Atelectasis":        0.47,
    "Effusion":           0.43,
    "Fibrosis":           0.50,
    "Pleural_Thickening": 0.50,
    "Hernia":             0.45,
    "No Finding":         0.50,   # never used — No Finding never fires GradCAM
}
 
# Color per condition (BGR for cv2)
CONDITION_COLORS_BGR = {
    "Atelectasis":        (80,  200, 255),   # amber
    "Cardiomegaly":       (80,  255, 160),   # green
    "Effusion":           (255, 160, 80),    # blue
    "Infiltration":       (255, 255, 100),   # yellow
    "Mass":               (255, 80,  255),   # magenta
    "Nodule":             (100, 255, 255),   # cyan
    "Pneumonia":          (80,  80,  255),   # red
    "Pneumothorax":       (180, 255, 80),    # lime
    "Consolidation":      (255, 180, 80),    # light blue
    "Edema":              (200, 80,  200),   # purple

    "Pleural_Thickening": (255, 200, 130),   # sky

}
 
 
def overlay_heatmap(bgr_img, cam, alpha=0.45):
    """
    Returns: heatmap overlay (BGR), and the raw cam_r channel (for contour extraction).
    cam: [0,1] float array, same H×W as bgr_img
    """
    heat_uint8 = (cam * 255).astype(np.uint8)
    heatmap    = cv2.applyColorMap(heat_uint8, cv2.COLORMAP_JET)  # BGR
    overlay    = cv2.addWeighted(bgr_img, 1 - alpha, heatmap, alpha, 0)
    # Return the red channel of JET heatmap for contour thresholding
    cam_r = heatmap[:, :, 2]  # red channel encodes high-activation region in JET
    return overlay, cam_r
 
 
def draw_contours_and_bbox(bgr_img, cam_r, label, cam_threshold):
    """
    Draws filled contours + bounding box on a copy of bgr_img.
    cam_threshold: fraction of 255 (e.g. 0.40 → thresh=102)
    """
    result   = bgr_img.copy()
    color    = CONDITION_COLORS_BGR.get(label, (255, 255, 255))
    thresh_v = int(cam_threshold * 255)
 
    _, binary = cv2.threshold(cam_r, thresh_v, 255, cv2.THRESH_BINARY)
    # Morphological cleanup
    kernel  = np.ones((5, 5), np.uint8)
    binary  = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, kernel)
    binary  = cv2.morphologyEx(binary, cv2.MORPH_OPEN,  kernel)
 
    contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
 
    # Filter noise contours (min area = 0.5% of image area)
    h, w    = bgr_img.shape[:2]
    min_area = 0.005 * h * w
    contours = [c for c in contours if cv2.contourArea(c) > min_area]
 
    cv2.drawContours(result, contours, -1, color, 2)
 
    for c in contours:
        x, y, cw, ch = cv2.boundingRect(c)
        cv2.rectangle(result, (x, y), (x + cw, y + ch), color, 2)
        cv2.putText(result, label.replace("_", " "),
                    (x, max(y - 6, 12)),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.45, color, 1, cv2.LINE_AA)
 
    return result, contours
 
 
def denorm_to_bgr(img_tensor):
    """Convert normalised CHW tensor to uint8 BGR for cv2."""
    mean = np.array([0.485, 0.456, 0.406])
    std  = np.array([0.229, 0.224, 0.225])
    img  = img_tensor.permute(1, 2, 0).cpu().numpy()
    img  = img * std + mean
    img  = np.clip(img, 0, 1)
    img_rgb  = (img * 255).astype(np.uint8)
    img_bgr  = cv2.cvtColor(img_rgb, cv2.COLOR_RGB2BGR)
    return img_bgr

    bgr_orig = denorm_to_bgr(img_tensor)
 
    with torch.no_grad():
        logits = model(input_t)
        probs_i = torch.sigmoid(logits)[0].cpu().numpy()
 
    active_pred = [label_cols[i] for i, p in enumerate(probs_i)
                   if p >= final_thresholds[label_cols[i]] and label_cols[i] != "No Finding"]
 
    # Skip pure No Finding predictions for localization

        continue
 
    # Build 4-panel figure: original | heatmap blend | contours+bbox | gt labels
    all_cams    = {}
    overlay_acc = bgr_orig.copy()   # accumulate all conditions on one overlay
 
    for label in active_pred:
        class_idx   = label_cols.index(label)
        cam_up      = gradcam.generate_upsampled(input_t, class_idx)
        all_cams[label] = cam_up
 
        # Update stats
        gt_pos = int(target[class_idx].item() > 0.5)
        localization_stats[label]["total_pred_pos"] += 1
        localization_stats[label]["total_gt_pos"]   += gt_pos
 
    # Build combined overlay (blend all active CAMs)
    if all_cams:
        combined_cam = np.max(np.stack(list(all_cams.values())), axis=0)
        heatmap_overlay, _ = overlay_heatmap(bgr_orig.copy(), combined_cam, alpha=0.5)
 
        # Draw per-condition contours
        contour_img = bgr_orig.copy()
        for label, cam_up in all_cams.items():
            _, cam_r_i = overlay_heatmap(bgr_orig.copy(), cam_up, alpha=0.0)
            contour_img, cnts = draw_contours_and_bbox(contour_img, cam_r_i,
                                                        label, CAM_THRESHOLDS[label])
            if cnts:
                localization_stats[label]["has_contour"]  += 1
            else:
                localization_stats[label]["no_contour"]   += 1
 
        if SAVE_LOCALIZATION:
            # Save 3-panel: original | heatmap | contours
            panel = np.hstack([bgr_orig, heatmap_overlay, contour_img])
            gt_names   = [label_cols[i].replace("_", " ")
                          for i, v in enumerate(target.numpy()) if v > 0.5]
